async function fetchSymbols(syms) {
  const url = `https://query1.finance.yahoo.com/v7/finance/quote?symbols=${syms.join(',')}&fields=regularMarketPrice,regularMarketChangePercent,regularMarketChange,regularMarketPreviousClose,marketCap,regularMarketVolume`;
  try {
    const r = await fetch(PROXY + encodeURIComponent(url), {signal: AbortSignal.timeout(12000)});
    const d = await r.json();
    return d?.quoteResponse?.result || [];
  } catch(e) { return []; }
}

// ── FORMAT ──
const f = (n,d=2) => n != null ? Number(n).toLocaleString('en-US',{minimumFractionDigits:d,maximumFractionDigits:d}) : '—';
const fc = n => (n>=0?'+':'')+f(n)+'%';

const mcap = v => !v ? '—' : v>1e12 ? '$'+f(v/1e12,2)+'T' : v>1e9 ? '$'+f(v/1e9,1)+'B' : '$'+f(v/1e6,0)+'M';

function buildTicker() {
  const items = [
    ...INDICES.map(i=>({sym:i.name.replace(' ','').slice(0,5), d:mkt[i.sym]})),
    ...STOCKS.map(s=>({sym:s.sym, d:mkt[s.sym]})),
    ...CRYPTOS.map(c=>({sym:c.label, d:mkt[c.sym]})),
    ...FOREX.map(x=>({sym:x.label, d:mkt[x.sym]})),
  ].filter(i=>i.d?.regularMarketPrice);

  if(!items.length) return;
  const html = items.map(i=>{
    const p = i.d.regularMarketPrice;
    const chg = i.d.regularMarketChangePercent||0;
    return `<span class="t-item"><span class="t-sym">${i.sym}</span><span class="t-val">${f(p)}</span><span class="t-chg ${bCls(chg)}">${fc(chg)}</span></span>`;
  }).join('');
  document.getElementById('tk').innerHTML = html+html;
}

// ── PORTFOLIO ──
function addHolding() {
  const sym = document.getElementById('p-sym').value.toUpperCase().trim();
  const qty = parseFloat(document.getElementById('p-qty').value);
  const cost = parseFloat(document.getElementById('p-cost').value);
  if(!sym||!qty||isNaN(qty)||!cost||isNaN(cost)) return;
  portfolio.push({sym,qty,cost,id:Date.now()});
  localStorage.setItem('fp2', JSON.stringify(portfolio));
  ['p-sym','p-qty','p-cost'].forEach(id => document.getElementById(id).value='');

function renderPortfolio() {
  const list = document.getElementById('port-list');
  const totEl = document.getElementById('port-total');
  if(!list) return;
  if(!portfolio.length) {
    list.innerHTML = '<div style="color:var(--t3);font-size:11px;padding:8px">No holdings yet. Add below.</div>';
    totEl.innerHTML = 'Add holdings below to track live P&L.';
    return;
  }
  let tCost=0, tVal=0;
  list.innerHTML = portfolio.map(h => {
    const key = [h.sym, h.sym+'-USD', h.sym+'=X'].find(k=>mkt[k]);
    const d = key ? mkt[key] : null;
    const price = d?.regularMarketPrice;
    const val = price ? price*h.qty : null;
    const pnl = val !== null ? val-h.cost : null;
    const pct = pnl!==null && h.cost ? (pnl/h.cost*100) : null;
    if(val!==null){tVal+=val;tCost+=h.cost;}
    const pnlColor = pnl===null?'var(--t3)':pnl>=0?'var(--gain)':'var(--loss)';
    return `<div class="p-entry">
      <div>
        <div class="p-sym">${h.sym} <span style="color:var(--t3);font-size:9px;font-weight:400">× ${h.qty}</span></div>
        <div class="p-detail">Cost $${f(h.cost)}</div>
      </div>
      <div>
        <div class="p-val">${val!==null?'$'+f(val):'—'}</div>
        <div class="p-pnl" style="color:${pnlColor}">${pnl!==null?(pnl>=0?'+':'')+' $'+f(Math.abs(pnl))+' ('+f(Math.abs(pct),1)+'%)':'no price'}</div>
      </div>
    </div>`;
  }).join('');
  if(tVal) {
    const tpnl=tVal-tCost, tpct=tCost?(tpnl/tCost*100):0;
    const c = tpnl>=0?'var(--gain)':'var(--loss)';
    totEl.innerHTML=`<div style="display:flex;justify-content:space-between;margin-bottom:3px"><span>Portfolio value</span><b>$${f(tVal)}</b></div>
    <div style="display:flex;justify-content:space-between"><span>Total P&L</span><b style="color:${c}">${tpnl>=0?'+':''}$${f(tpnl)} (${f(tpct,1)}%)</b></div>`;
  }
}

async function send() {
  const el = document.getElementById('chat-in');
  const txt = el.value.trim();
  if(!txt) return;
  el.value=''; el.style.height='auto';
  addMsg(txt,'u');
  chatHistory.push({role:'user',content:txt});
  document.getElementById('send-btn').disabled=true;
  const t = showTyping();
  try {
    const res = await fetch('https://api.anthropic.com/v1/messages',{
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body:JSON.stringify({
        model:'claude-sonnet-4-6',
        max_tokens:1000,
        system:sysPrompt(),
        messages:chatHistory
      })
    });
    t.remove();
    const data = await res.json();
    const reply = data.content?.[0]?.text || 'Sorry, connection issue. Try again.';
    chatHistory.push({role:'assistant',content:reply});
    addMsg(fmt(reply),'a');
  } catch(e) {
    t.remove();
    addMsg('Network error — check connection and try again.','a');
  }
  document.getElementById('send-btn').disabled=false;
}

function fmt(t) {
  return t.replace(/\*\*(.*?)\*\*/g,'<strong>$1</strong>').replace(/\n\n/g,'<br><br>').replace(/\n/g,'<br>');
}

function showTyping() {
  const d = document.createElement('div');
  d.className='typing';
  d.innerHTML='<span></span><span></span><span></span>';
  document.getElementById('msgs').appendChild(d);
  d.scrollIntoView({behavior:'smooth'});
  return d;
}

function addMsg(text, role) {
  const msgs = document.getElementById('msgs');
  const d = document.createElement('div');
  d.className='msg msg-'+role;
  const now = new Date().toLocaleTimeString([],{hour:'2-digit',minute:'2-digit'});
  d.innerHTML = (role==='a' ? text : esc(text)) + `<div class="msg-ts">${now}</div>`;
  msgs.appendChild(d);
  d.scrollIntoView({behavior:'smooth'});
}

function esc(t){ return t.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;'); }

function qa(q) {
  document.getElementById('chat-in').value=q;
  send();
}

function onKey(e) {
  if(e.key==='Enter'&&!e.shiftKey){e.preventDefault();send();}
}

function grow(el) {
  el.style.height='auto';
  el.style.height=Math.min(el.scrollHeight,80)+'px';
}
